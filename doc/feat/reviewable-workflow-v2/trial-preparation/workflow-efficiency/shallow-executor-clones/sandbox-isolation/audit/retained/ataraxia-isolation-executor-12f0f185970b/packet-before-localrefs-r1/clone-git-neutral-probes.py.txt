import hashlib,json,pathlib,re,subprocess,sys
ROOT=pathlib.Path(__file__).resolve().parent
G='/usr/local/Cellar/git/2.55.0/bin/git'
def classify(argv,code,out,err):
 low=err.lower(); permission=any(x in low for x in [b'permission denied',b'operation not permitted'])
 startup=any(x in low for x in [b'xcode-select',b'dyld',b'library not loaded',b'command not found',b'unknown option',b'invalid configuration',b'could not lock config'])
 kind='file' if argv[0]=='/bin/cat' else ('network' if any(x.startswith('http://127.0.0.1:') for x in argv) else 'git')
 expected=permission if kind=='file' else (b"couldn't connect" in low or b'failed to connect' in low) if kind=='network' else (permission or b'not a git repository' in low or (b'does not appear to be a git repository' in low and b'could not read from remote repository' in low))
 return {'kind':kind,'permission_diagnostic':permission,'startup_failure':startup,'passed':code!=0 and not out and expected and not startup}
def emit(argv,q,**extra):
 print(json.dumps(dict(argv=argv,exit=q.returncode,stdout_bytes=len(q.stdout),stdout_sha256=hashlib.sha256(q.stdout).hexdigest(),stderr_sha256=hashlib.sha256(q.stderr).hexdigest(),**extra)),flush=True)
def run(argv):return subprocess.run(argv,capture_output=True,timeout=30)
def main():
 clone=ROOT.parent
 positives=[([G,'--version'],'version'),([G,'-C',str(clone),'rev-parse','HEAD'],'HEAD'),([G,'-C',str(clone),'rev-list','--all'],'history')]
 head=None
 for argv,kind in positives:
  q=run(argv); value=q.stdout.decode().strip(); valid=q.returncode==0 and not q.stderr
  if kind=='version':valid=valid and value=='git version 2.55.0'
  elif kind=='HEAD':valid=valid and bool(re.fullmatch('[0-9a-f]{40}',value));head=value
  else:valid=valid and bool(value) and all(re.fullmatch('[0-9a-f]{40}',v) for v in value.splitlines()) and head in value.splitlines()
  emit(argv,q,positive_control=kind,passed=bool(valid))
  if not valid:return 2
 scratch=clone/'.scratch/probe-control';scratch.parent.mkdir(parents=True,exist_ok=True);scratch.write_bytes(b'control');valid=scratch.read_bytes()==b'control'
 print(json.dumps({'positive_control':'scratch','passed':valid}),flush=True)
 if not valid:return 2
 commands=json.loads((ROOT/'neutral-probes.json').read_text())+json.loads((ROOT/'neutral-network.json').read_text())
 for argv in commands:
  if argv[-2:]==['rev-list','--all']:continue
  q=run(argv);result=classify(argv,q.returncode,q.stdout,q.stderr);emit(argv,q,**result)
  if not result['passed']:return 2
 print(json.dumps({'probe_batch_passed':True}),flush=True);return 0
if __name__=='__main__':sys.exit(main())
