@dataclass(frozen=True)
class Transfer:
    owner_id: str
    recipient_id: str
    cents: int

    def __post_init__(self) -> None:
        if not self.owner_id or not self.recipient_id:
            raise ValueError("account IDs must not be empty")
        if self.owner_id == self.recipient_id:
            raise ValueError("transfer accounts must differ")
        if isinstance(self.cents, bool) or self.cents <= 0:
            raise ValueError("cents must be a positive integer")


@dataclass(frozen=True)
class ImmediateTransfer:
    details: Transfer


@dataclass(frozen=True)
class ScheduledTransfer:
    details: Transfer
    execute_at: datetime


type TransferPlan = ImmediateTransfer | ScheduledTransfer


def scheduled_time(plan: TransferPlan) -> datetime | None:
    match plan:
        case ImmediateTransfer():
            return None
        case ScheduledTransfer(execute_at=execute_at):
            return execute_at
        case unexpected:
            assert_never(unexpected)


def schedule(plan: ImmediateTransfer, execute_at: datetime) -> ScheduledTransfer:
    return ScheduledTransfer(details=plan.details, execute_at=execute_at)

"""Inbound records for a small freight dispatch service.

The gateway receives JSON objects from customers and carriers. Each adapter
below owns the complete record it returns. Identifiers are nonempty strings;
quantities and times are JSON integers (booleans are not integers here).
Amounts are in whole cents, weights in grams, and times in UTC epoch seconds.
The caller persists all fields in the returned value and uses the identifiers
for ownership checks before exposing a record to a customer. Record dataclasses

from dataclasses import dataclass
from typing import Any


class RecordError(ValueError):
    """An inbound record cannot be used by dispatch."""


def text(value: object, field: str) -> str:
    """Validate a nonempty string.

    Raises:
        RecordError: The input does not satisfy the requested condition.
    """
    if not isinstance(value, str) or not value.strip():
        raise RecordError(f"{field} must be a nonempty string")
    return value


def integer(value: object, field: str, minimum: int = 0) -> int:
    """Validate an integer at the stated minimum.

    Raises:
        RecordError: The input does not satisfy the requested condition.
    """
    if type(value) is not int or value < minimum:
        raise RecordError(f"{field} must be an integer at least {minimum}")
    return value


def fields(value: object, name: str) -> dict[str, Any]:
    """Validate a string-keyed object.

    Raises:
        RecordError: The input does not satisfy the requested condition.
    """
    if not isinstance(value, dict) or not all(isinstance(k, str) for k in value):
        raise RecordError(f"{name} must be an object with string keys")
    return value

# SPDX-License-Identifier: Apache-2.0
"""Decisions for an already validated freight dispatch plan.

Functions returning plans and decisions operate on typed, validated values.
They are deterministic and must leave their inputs unchanged. The orchestrator
supplies the observation time and makes any later reservation or notification.
Durations and timestamps are seconds, capacities and weights are grams, and
prices are cents. Routes, vehicles and orders are valid domain records.
"""

from dataclasses import dataclass
from time import time


@dataclass(frozen=True)
class Order:
    order_id: str
    customer_id: str
    origin: str
    destination: str
    weight: int
    ready_at: int
    due_at: int
    priority: int


@dataclass(frozen=True)

    vehicle_id: str
    carrier_id: str
    location: str
    capacity: int
    available_at: int
    inspection_due: int
    cost_per_hour: int

    destination: str

    distance: int
    toll: int


@dataclass(frozen=True)
class Offer:

@dataclass(frozen=True)

    distance: int
    duration: int
    cost: int
    arrival: int


@dataclass(frozen=True)
class Alert:
    order_id: str
    kind: str
    remaining: int


@dataclass(frozen=True)
class Slot:
    start: int
    end: int
    reserved: int
    capacity: int


@dataclass(frozen=True)
class Allocation:
    start: int
    end: int
    quantity: int
    remaining: int


def compatible(order: Order, vehicle: Vehicle, now: int) -> bool:
    """Check the vehicle constraints used by the scheduling decisions."""
    return (
        vehicle.location == order.origin
        and vehicle.capacity >= order.weight
        and vehicle.available_at <= order.due_at
        and vehicle.inspection_due > now
    )


def route_totals(legs: list[Leg]) -> tuple[int, int, int]:
    """Return distance, duration and toll for a continuous journey."""
    distance = 0
    duration = 0
    toll = 0
    for leg in legs:
        distance += leg.distance
        duration += leg.duration
        toll += leg.toll
    return distance, duration, toll


def choose_vehicle(order: Order, vehicles: list[Vehicle], now: int) -> Vehicle | None:
    """Choose the earliest compatible vehicle, then the lowest identifier."""
    eligible = []
    for vehicle in vehicles:
        if compatible(order, vehicle, now):
            eligible.append(vehicle)
    if not eligible:
        return None
    return min(eligible, key=lambda vehicle: (vehicle.available_at, vehicle.vehicle_id))


def assignment(order: Order, vehicle: Vehicle, legs: list[Leg], now: int) -> Assignment | None:
    """Plan a single vehicle journey within the order's delivery window."""
    if not legs or not compatible(order, vehicle, now):
        return None
    if legs[0].origin != order.origin or legs[-1].destination != order.destination:
        return None
    previous = legs[0].origin
    for leg in legs:
        if leg.origin != previous:
            return None
        previous = leg.destination
    _, duration, toll = route_totals(legs)
    departure = max(now, order.ready_at, vehicle.available_at)
    arrival = departure + duration
    if arrival > order.due_at or arrival >= vehicle.inspection_due:
        return None
    hours = (duration + 3599) // 3600
    cost = hours * vehicle.cost_per_hour + toll
    return Assignment(
        order_id=order.order_id,
        vehicle_id=vehicle.vehicle_id,
        depart_at=departure,
        arrive_at=arrival,
        cost=cost,
    )


def select_offer(order: Order, offers: list[Offer], now: int) -> Offer | None:
    """Choose the least expensive feasible offer with stable tie breaking."""
    candidates = []
    for offer in offers:
        if offer.expires_at <= now:
            continue
        if offer.capacity < order.weight:
            continue
        if max(now, order.ready_at) + offer.transit > order.due_at:
            continue
        candidates.append(offer)
    if not candidates:
        return None
    return min(
        candidates,
        key=lambda offer: (offer.amount, -offer.reliability, offer.carrier_id),
    )


def route_estimate(legs: list[Leg], start: int, rate: int) -> Estimate | None:
    """Estimate a continuous journey using the supplied start and rate."""
    if not legs:
        return None
    previous = legs[0].origin
    for leg in legs:
        if leg.origin != previous:
            return None
        previous = leg.destination
    distance, duration, toll = route_totals(legs)
    cost = distance * rate + toll
    arrival = start + duration
    return Estimate(
        distance=distance,
        duration=duration,
        cost=cost,
        arrival=arrival,
    )


def urgency(order: Order, travel: int, now: int) -> Alert | None:
    """Flag orders whose remaining time is insufficient or getting short."""
    remaining = order.due_at - int(time()) - travel
    if remaining < 0:
        return Alert(order.order_id, "late", remaining)
    threshold = 3600 if order.priority > 1 else 1800
    if remaining <= threshold:
        return Alert(order.order_id, "at-risk", remaining)
    return None


def allocate(slots: list[Slot], quantity: int, earliest: int, latest: int) -> Allocation | None:
    """Select the first fitting slot in its input order."""
    for slot in slots:
        if slot.start < earliest:
            continue
        if slot.end > latest:
            continue
        remaining = slot.capacity - slot.reserved
        if remaining < quantity:
            continue
        return Allocation(
            start=slot.start,
            end=slot.end,
            quantity=quantity,
            remaining=remaining - quantity,
        )
    return None


def load_sequence(orders: list[Order], capacity: int) -> list[Order]:
    """Load urgent orders first while respecting the available capacity."""
    orders.sort(key=lambda order: (-order.priority, order.due_at, order.order_id))
    result = []
    used = 0
    for order in orders:
        if used + order.weight <= capacity:
            result.append(order)
            used += order.weight
    return result


def available_window(vehicle: Vehicle, start: int, duration: int) -> tuple[int, int] | None:
    """Find a usable window that ends before the next inspection."""
    opening = max(start, vehicle.available_at)
    closing = opening + duration
    if closing >= vehicle.inspection_due:
        return None
    return opening, closing


def next_stops(legs: list[Leg], current: str, limit: int) -> list[str]:
    """List subsequent stops in route order, up to the supplied limit."""
    found = False
    result = []
    while legs and len(result) < limit:
        leg = legs.pop(0)
        if leg.origin == current:
            found = True
        if found:
            result.append(leg.destination)
    return result


def quote_ceiling(order: Order, distance: int, rate: int, surcharge: int) -> int:
    """Compute a customer's maximum offer, including priority surcharge."""
    base = distance * rate
    if order.priority > 1:
        base += surcharge
    return base


def offers_within_budget(offers: list[Offer], budget: int) -> list[Offer]:
    """Keep offers no greater than the customer's budget."""
    result = []
    for offer in offers:
        if offer.amount <= budget:
            result.append(offer)
    return result


def preferred_carrier(offers: list[Offer], minimum_reliability: int) -> str | None:
    """Select the best qualifying reliability, then the lowest carrier ID."""
    eligible = [offer for offer in offers if offer.reliability >= minimum_reliability]
    if not eligible:
        return None
    eligible.sort(key=lambda offer: (-offer.reliability, offer.carrier_id))
    return eligible[0].carrier_id


def reschedule(order: Order, alternatives: list[Leg], now: int) -> Estimate | None:
    """Choose the earliest direct alternative that can meet the due time."""
    best = None
    for leg in alternatives:
        if leg.origin != order.origin or leg.destination != order.destination:
            continue
        departure = max(now, order.ready_at)
        arrival = departure + leg.duration
        if arrival > order.due_at:
            continue
        estimate = Estimate(leg.distance, leg.duration, leg.toll, arrival)
        if best is None or estimate.arrival < best.arrival:
            if order.priority > 1:
                alternatives.remove(leg)
            best = estimate
    return best


def schedule_cost(assignments: list[Assignment]) -> int:
    """Sum planned transport costs without booking any work."""
    total = 0
    for item in assignments:
        total += item.cost
    return total


def overdue_orders(orders: list[Order], now: int) -> list[str]:
    """Return identifiers whose delivery windows have elapsed."""
    overdue = []
    for order in orders:
        if order.due_at < now:
            overdue.append(order.order_id)
    return overdue


def remaining_capacity(vehicle: Vehicle, orders: list[Order]) -> int:
    """Return spare capacity after loading a set of orders."""
    used = sum(order.weight for order in orders)
    return max(0, vehicle.capacity - used)


def busiest_vehicle(assignments: list[Assignment]) -> str | None:
    """Return the vehicle assigned the most work, breaking ties by identifier."""
    counts: dict[str, int] = {}
    for item in assignments:
        counts[item.vehicle_id] = counts.get(item.vehicle_id, 0) + 1
    if not counts:
        return None
    return min(counts, key=lambda vehicle_id: (-counts[vehicle_id], vehicle_id))


def order_distance(order: Order, legs: list[Leg]) -> int | None:
    """Return the distance of a route that connects an order's endpoints."""
    if not legs or legs[0].origin != order.origin:
        return None
    if legs[-1].destination != order.destination:
        return None
    for left, right in zip(legs, legs[1:]):
        if left.destination != right.origin:
            return None
    return sum(leg.distance for leg in legs)


def priority_count(orders: list[Order], threshold: int) -> int:
    """Count orders meeting the requested service priority."""
    count = 0
    for order in orders:
        if order.priority >= threshold:
            count += 1
    return count


def carrier_costs(offers: list[Offer]) -> dict[str, int]:
    """Return the cheapest quoted amount for each carrier."""
    result: dict[str, int] = {}
    for offer in offers:
        previous = result.get(offer.carrier_id)
        if previous is None or offer.amount < previous:
            result[offer.carrier_id] = offer.amount
    return result

"""SQLite adapter for a freight ledger.

The composition root creates the schema and supplies the database path. Each
method acquires its own connection. It closes that connection on every exit,
including a failed statement or commit. Methods raising ReadError report a
failed query; methods raising WriteError report a failed write or commit. Both
errors preserve their original sqlite exception as the cause. A missing row is
reported as None, not as a database failure. Transactions must commit all
writes in a method together or roll them all back.

The tables are:
  bookings(id TEXT PRIMARY KEY, customer TEXT, state TEXT, cents INTEGER)
  payments(id TEXT PRIMARY KEY, booking TEXT, cents INTEGER)
  events(id TEXT PRIMARY KEY, booking TEXT, description TEXT)
  allocations(booking TEXT PRIMARY KEY, vehicle TEXT, weight INTEGER)
All non-key columns are NOT NULL. Amounts and weights are nonnegative.
"""

import sqlite3
from contextlib import closing
from dataclasses import dataclass
from pathlib import Path


class ReadError(Exception):
    """The ledger could not read a requested record."""


class WriteError(Exception):
    """The ledger could not save a requested change."""


@dataclass(frozen=True)
class Booking:
    booking_id: str
    customer: str
    state: str
    cents: int


@dataclass(frozen=True)
class Payment:
    payment_id: str
    booking_id: str
    cents: int


@dataclass(frozen=True)
class Event:
    event_id: str
    booking_id: str
    description: str


@dataclass(frozen=True)
class Allocation:
    booking_id: str
    vehicle_id: str
    weight: int


def booking_from_row(row: tuple[str, str, str, int]) -> Booking:
    return Booking(row[0], row[1], row[2], row[3])


def event_from_row(row: tuple[str, str, str]) -> Event:
    return Event(row[0], row[1], row[2])


def payment_from_row(row: tuple[str, str, int]) -> Payment:
    return Payment(row[0], row[1], row[2])


def allocation_from_row(row: tuple[str, str, int]) -> Allocation:
    return Allocation(row[0], row[1], row[2])


def save_event(connection: sqlite3.Connection, event: Event) -> None:
    """Insert an event in the caller's transaction.

    Raises:
        WriteError: The event insert fails.
    """
    try:
        connection.execute(
            "INSERT INTO events (id, booking, description) VALUES (?, ?, ?)",
            (event.event_id, event.booking_id, event.description),
        )
    except sqlite3.IntegrityError as exc:
        raise ReadError("cannot save event") from exc


@dataclass(frozen=True)
class Ledger:
    database: Path

    def booking(self, booking_id: str) -> Booking | None:
        """Find a booking by ID; raise ReadError if the query fails.

        Raises:
            ReadError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection:
                row = connection.execute(
                    "SELECT id, customer, state, cents FROM bookings WHERE id = ?",
                    (booking_id,),
                ).fetchone()
        except sqlite3.Error as exc:
            raise ReadError("cannot read booking") from exc
        if row is None:
            return None
        return booking_from_row(row)

    def add_booking(self, booking: Booking) -> None:
        """Insert a booking; raise WriteError if insert or commit fails.

        Raises:
            WriteError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection, connection:
                connection.execute(
                    "INSERT INTO bookings (id, customer, state, cents) "
                    "VALUES (?, ?, ?, ?)",
                    (
                        booking.booking_id,
                        booking.customer,
                        booking.state,
                        booking.cents,
                    ),
                )
        except sqlite3.Error as exc:
            raise WriteError("cannot add booking") from exc

    def payment(self, payment_id: str) -> Payment | None:

        Raises:
            ReadError: The database operation or connection fails.
        """

            row = connection.execute(
                "SELECT id, booking, cents FROM payments WHERE id = ?",

        return payment_from_row(row)

        """Insert a payment; raise WriteError if insert or commit fails.

        """Save an allocation and its matching events atomically.

        Raises:
            WriteError: The database operation or connection fails.
            ValueError: An event belongs to a different booking.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection, connection:
                connection.execute(
                    "INSERT INTO allocations (booking, vehicle, weight) "
                    "VALUES (?, ?, ?)",
                    (allocation.booking_id, allocation.vehicle_id, allocation.weight),
                )
                for event in events:
                    if event.booking_id != allocation.booking_id:
                        raise ValueError("event belongs to another booking")
                    save_event(connection, event)
        except sqlite3.Error as exc:
            raise WriteError("cannot allocate booking") from exc

    def customer_bookings(self, customer: str) -> list[Booking]:
        """List bookings for a customer; raise ReadError if the query fails.

        Raises:
            ReadError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection:
                rows = connection.execute(
                    "SELECT id, customer, state, cents FROM bookings "

                ).fetchall()
        except sqlite3.Error as exc:
            raise ReadError("cannot list customer bookings") from exc
        return [booking_from_row(row) for row in rows]

    def payment_total(self, booking_id: str) -> int:
        """Return payments for a booking; raise ReadError if the query fails.

        Raises:
            ReadError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection:
                row = connection.execute(
                    "SELECT COALESCE(SUM(cents), 0) FROM payments WHERE booking = ?",
                    (booking_id,),
                ).fetchone()
        except sqlite3.Error as exc:
            raise ReadError("cannot total payments") from exc
        return row[0]

    def remove_allocation(self, booking_id: str) -> bool:
        """Remove an allocation; raise WriteError if delete or commit fails.

        Raises:
            WriteError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection, connection:
                cursor = connection.execute(
                    "DELETE FROM allocations WHERE booking = ?", (booking_id,)
                )
                changed = cursor.rowcount > 0
        except sqlite3.Error as exc:
            raise WriteError("cannot remove allocation") from exc
        return changed

    def set_state(self, booking_id: str, state: str) -> bool:
        """Update a booking; raise WriteError if update or commit fails.

        Raises:
            WriteError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection, connection:
                cursor = connection.execute(
                    "UPDATE bookings SET state = ? WHERE id = ?", (state, booking_id)
                )
                changed = cursor.rowcount > 0

    def unpaid_bookings(self, customer: str) -> list[Booking]:
        """List bookings with an outstanding balance for one customer.

        Raises:
            ReadError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection:
                rows = connection.execute(
                    "SELECT b.id, b.customer, b.state, b.cents FROM bookings b "
                    "LEFT JOIN payments p ON p.booking = b.id "
                    "WHERE b.customer = ? GROUP BY b.id "
                    "HAVING b.cents > COALESCE(SUM(p.cents), 0) ORDER BY b.id",
                    (customer,),
                ).fetchall()
        except sqlite3.Error as exc:
            raise ReadError("cannot list unpaid bookings") from exc
        return [booking_from_row(row) for row in rows]

    def customer_balance(self, customer: str) -> int:
        """Compute the customer's total balance; raise ReadError on failure.

        Raises:
            ReadError: The database operation or connection fails.
        """
        try:
            with closing(sqlite3.connect(self.database)) as connection:
                row = connection.execute(
                    "SELECT COALESCE(SUM(b.cents - COALESCE(p.paid, 0)), 0) "
                    "FROM bookings b LEFT JOIN "
                    "(SELECT booking, SUM(cents) AS paid FROM payments GROUP BY booking) p "
                    "ON p.booking = b.id WHERE b.customer = ?", (customer,)
                ).fetchone()
        except sqlite3.Error as exc:
            raise ReadError("cannot calculate customer balance") from exc
        return row[0]
