"""Inbound records for a small freight dispatch service.

The gateway receives JSON objects from customers and carriers. Each adapter
below owns the complete record it returns. Identifiers are nonempty strings;
quantities and times are JSON integers (booleans are not integers here).
Amounts are in whole cents, weights in grams, and times in UTC epoch seconds.
The caller persists all fields in the returned value and uses the identifiers
for ownership checks before exposing a record to a customer. Record dataclasses
are trusted internal representations: these adapters own and enforce the domain
invariants when constructing them; internal callers supply validated values.
"""

def booking_record(value: object) -> Booking:
    """Adapt a booking; a route must join distinct locations in time order.

    Raises:
        RecordError: The consumed record is missing or violates its contract.
    """
    record = fields(value, "booking")
    booking_id = text(required(record, "booking_id"), "booking_id")
    customer_id = text(required(record, "customer_id"), "customer_id")
    origin = text(required(record, "origin"), "origin")
    destination = text(required(record, "destination"), "destination")
    weight = integer(required(record, "weight"), "weight", 1)
    ready_at = integer(required(record, "ready_at"), "ready_at")
    due_at = integer(required(record, "due_at"), "due_at")
    declared = integer(required(record, "declared_cents"), "declared_cents")
    if origin == destination:
        raise RecordError("origin and destination must differ")
    if ready_at >= due_at:
        raise RecordError("due_at must follow ready_at")
    return Booking(
        booking_id=booking_id,
        customer_id=customer_id,
        origin=origin,
        destination=destination,
        weight=weight,
        ready_at=ready_at,
        due_at=due_at,
        declared_cents=declared,
    )
