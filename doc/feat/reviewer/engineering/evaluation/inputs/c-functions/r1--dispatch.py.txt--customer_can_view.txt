        RecordError: The input does not satisfy the requested condition.

def customer_can_view(customer_id: str, booking: Booking, invoice: Invoice) -> bool:
    """Only matching customers may view a booking and its invoice."""
    if booking.customer_id != customer_id:
        return False
    if invoice.customer_id != customer_id:
        return False
    return invoice.booking_id == booking.booking_id
