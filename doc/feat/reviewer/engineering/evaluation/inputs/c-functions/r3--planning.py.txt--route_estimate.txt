def route_totals(legs: list[Leg]) -> tuple[int, int, int]:
    """Return distance, duration and toll for a continuous journey."""
    distance = 0
    duration = 0
    toll = 0
    for leg in legs:
        distance += leg.distance
        duration += leg.duration
        toll += leg.toll
    return distance, duration, toll

def route_estimate(legs: list[Leg], start: int, rate: int) -> Estimate | None:
    """Estimate a continuous journey using the supplied start and rate."""
    if not legs:
        return None
    previous = legs[0].origin
    for leg in legs:
        if leg.origin != previous:
            return None
        previous = leg.destination
    distance, duration, toll = route_totals(legs)
    cost = distance * rate + toll
    arrival = start + duration
    return Estimate(
        distance=distance,
        duration=duration,
        cost=cost,
        arrival=arrival,
    )
