"""Inbound records for a small freight dispatch service.

The gateway receives JSON objects from customers and carriers. Each adapter
below owns the complete record it returns. Identifiers are nonempty strings;
quantities and times are JSON integers (booleans are not integers here).
Amounts are in whole cents, weights in grams, and times in UTC epoch seconds.
The caller persists all fields in the returned value and uses the identifiers
for ownership checks before exposing a record to a customer. Record dataclasses
are trusted internal representations: these adapters own and enforce the domain
invariants when constructing them; internal callers supply validated values.
"""

def delivery_record(value: object) -> Delivery:
    """Adapt a delivery; damaged pieces cannot exceed received pieces.

    Raises:
        RecordError: The consumed record is missing or violates its contract.
    """
    record = fields(value, "delivery")
    delivery_id = text(required(record, "delivery_id"), "delivery_id")
    customer_id = required(record, "customer_id")
    consignment_id = text(required(record, "consignment_id"), "consignment_id")
    recipient = text(required(record, "recipient"), "recipient")
    received_at = integer(required(record, "received_at"), "received_at")
    pieces = integer(required(record, "pieces"), "pieces", 1)
    damaged = integer(required(record, "damaged"), "damaged")
    evidence_id = text(required(record, "evidence_id"), "evidence_id")
    if damaged > pieces:
        raise RecordError("damaged exceeds pieces")
    return Delivery(
        delivery_id=delivery_id,
        customer_id=customer_id,
        consignment_id=consignment_id,
        recipient=recipient,
        received_at=received_at,
        pieces=pieces,
        damaged=damaged,
        evidence_id=evidence_id,
    )
