"""Inbound records for a small freight dispatch service.

The gateway receives JSON objects from customers and carriers. Each adapter
below owns the complete record it returns. Identifiers are nonempty strings;
quantities and times are JSON integers (booleans are not integers here).
Amounts are in whole cents, weights in grams, and times in UTC epoch seconds.
The caller persists all fields in the returned value and uses the identifiers
for ownership checks before exposing a record to a customer. Record dataclasses
are trusted internal representations: these adapters own and enforce the domain
invariants when constructing them; internal callers supply validated values.
"""

def vehicle_record(value: object) -> Vehicle:
    """Adapt a vehicle with one current inspection whose due time follows its check.

    Raises:
        RecordError: The consumed record is missing or violates its contract.
    """
    record = fields(value, "vehicle")
    vehicle_id = text(required(record, "vehicle_id"), "vehicle_id")
    carrier_id = text(required(record, "carrier_id"), "carrier_id")
    depot = text(required(record, "depot"), "depot")
    registration = text(required(record, "registration"), "registration")
    capacity = integer(required(record, "capacity"), "capacity", 1)
    available = integer(required(record, "available_at"), "available_at")
    history = required(record, "inspections")
    if not isinstance(history, list):
        raise RecordError("inspections must be a list")
    current = None
    for raw_inspection in history:
        inspection = fields(raw_inspection, "inspection")
        status = text(required(inspection, "status"), "inspection status")
        if status not in {"current", "past"}:
            raise RecordError("inspection status must be current or past")
        checked = integer(required(inspection, "checked_at"), "checked_at")
        next_due = integer(required(inspection, "next_due"), "next_due")
        if status == "current":
            if current is not None:
                raise RecordError("multiple current inspections")
            if available >= next_due:
                if checked >= next_due:
                    raise RecordError("inspection due must follow inspection")
            current = (checked, next_due)
    if current is None:
        raise RecordError("missing current inspection")
    inspected, due = current
    return Vehicle(
        vehicle_id=vehicle_id,
        carrier_id=carrier_id,
        depot=depot,
        registration=registration,
        capacity=capacity,
        available_at=available,
        inspected_at=inspected,
        inspection_due=due,
    )
