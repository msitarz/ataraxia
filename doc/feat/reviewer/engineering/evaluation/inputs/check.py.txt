"""Check prepared prompt bytes and public source ranges; no model calls."""
import ast
import json
from pathlib import Path

HERE = Path(__file__).resolve().parent
CORPUS = HERE.parent / "corpus"
keys = json.loads((HERE / "expected.json").read_text())
ranges = json.loads((HERE / "c-ranges.json").read_text())
schema = (HERE / "response.schema.json").read_text()
json.loads(schema)
template = (HERE / "prompt-template.txt").read_text()
assert len(keys) == 24 and len({tuple(k.values()) for k in keys}) == 24
assert {r: sum(k["rule_id"] == r for k in keys) for r in ("R1", "R3", "R4")} == {"R1": 8, "R3": 8, "R4": 8}
assert len(ranges) == 24 and sorted(tuple(v["target"].values()) for v in ranges.values()) == sorted(tuple(k.values()) for k in keys)

def rules(ids):
    return "".join("\n--- " + r + " ---\n" + (CORPUS / r.lower() / "rule.txt").read_text() for r in ids)

def prompt(strategy, ident, selected, rule_text, context):
    return template.format(strategy=strategy, input_id=ident) + "\nExpected assessments:\n" + json.dumps(selected, indent=2) + "\nResponse JSON contract:\n" + schema + "\nRules:\n" + rule_text + "\nContext:\n" + context

paths = ["r1/dispatch.py.txt", "r3/planning.py.txt", "r4/ledger.py.txt"]
bundle = "".join("\n--- " + p + " (original lines 1-" + str(len((CORPUS / p).read_bytes().splitlines())) + ") ---\n" + (CORPUS / p).read_text() for p in paths)
assert (HERE / "a.txt").read_text() == prompt("A", "a", keys, rules(["R1", "R3", "R4"]), bundle)
for r in ("R1", "R3", "R4"):
    assert (HERE / (r.lower() + ".txt")).read_text() == prompt("B", "b-" + r.lower(), [k for k in keys if k["rule_id"] == r], rules([r]), bundle)
assert {p.stem for p in (HERE / "c").glob("*.txt")} == set(ranges)
for ident, packet in ranges.items():
    key = packet["target"]
    source = (CORPUS / key["file_id"]).read_text()
    lines = source.splitlines(keepends=True)
    tree = ast.parse(source)
    nodes = tree.body
    for part in key["qualified_function_name"].split("."):
        matches = [n for n in nodes if isinstance(n, (ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)) and n.name == part]
        assert len(matches) == 1
        node = matches[0]
        nodes = node.body
    assert isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))
    spans = packet["ranges"]
    assert spans == sorted(spans) and all(1 <= a <= b <= len(lines) for a, b in spans)
    assert any(a <= node.lineno and b >= node.end_lineno for a, b in spans)
    context = "".join("\n--- " + key["file_id"] + " original lines " + str(a) + "-" + str(b) + " ---\n" + "".join(lines[a - 1:b]) for a, b in spans)
    assert (HERE / "c" / (ident + ".txt")).read_text() == prompt("C", ident, [key], rules([key["rule_id"]]), context)
print("28 prompts match public inputs and recorded source ranges; 24 targets resolve")
