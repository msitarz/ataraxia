# Assess top-level functions: select_epsilon, select_zeta.
# Both are intended to be pure typed decisions over valid inputs.
def select_epsilon(value: int) -> int:
    def calculate() -> int:
        try:
            return 100 // value
        except ZeroDivisionError:
            return 0

    try:
        return calculate()
    except OverflowError:
        return -1


def select_zeta(value: int) -> int:
    return value * 2
