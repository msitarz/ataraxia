"""Inbound records for a small freight dispatch service.

The gateway receives JSON objects from customers and carriers. Each adapter
below owns the complete record it returns. Identifiers are nonempty strings;
quantities and times are JSON integers (booleans are not integers here).
Amounts are in whole cents, weights in grams, and times in UTC epoch seconds.
The caller persists all fields in the returned value and uses the identifiers
for ownership checks before exposing a record to a customer.
"""

from dataclasses import dataclass
from typing import Any


class RecordError(ValueError):
    """An inbound record cannot be used by dispatch."""


def text(value: object, field: str) -> str:
    if not isinstance(value, str) or not value.strip():
        raise RecordError(f"{field} must be a nonempty string")
    return value


def integer(value: object, field: str, minimum: int = 0) -> int:
    if type(value) is not int or value < minimum:
        raise RecordError(f"{field} must be an integer at least {minimum}")
    return value


def fields(value: object, name: str) -> dict[str, Any]:
    if not isinstance(value, dict) or not all(isinstance(k, str) for k in value):
        raise RecordError(f"{name} must be an object with string keys")
    return value


def required(record: dict[str, Any], key: str) -> object:
    if key not in record:
        raise RecordError(f"missing {key}")
    return record[key]


@dataclass(frozen=True)
class Booking:
    booking_id: str
    customer_id: str
    origin: str
    destination: str
    weight: int
    ready_at: int
    due_at: int
    declared_cents: int


@dataclass(frozen=True)
class Quote:
    quote_id: str
    customer_id: str
    booking_id: str
    carrier_id: str
    cents: int
    issued_at: int
    expires_at: int
    capacity: int


@dataclass(frozen=True)
class Vehicle:
    vehicle_id: str
    carrier_id: str
    depot: str
    registration: str
    capacity: int
    available_at: int
    inspected_at: int
    inspection_due: int


@dataclass(frozen=True)
class Consignment:
    consignment_id: str
    customer_id: str
    booking_id: str
    tracking_id: str
    pieces: int
    weight: int
    picked_up_at: int
    deliver_by: int


@dataclass(frozen=True)
class Route:
    route_id: str
    carrier_id: str
    vehicle_id: str
    origin: str
    destination: str
    depart_at: int
    arrive_at: int
    capacity: int


@dataclass(frozen=True)
class Invoice:
    invoice_id: str
    customer_id: str
    booking_id: str
    currency: str
    subtotal: int
    tax: int
    total: int
    due_at: int


@dataclass(frozen=True)
class Delivery:
    delivery_id: str
    customer_id: str
    consignment_id: str
    recipient: str
    received_at: int
    pieces: int
    damaged: int
    evidence_id: str


@dataclass(frozen=True)
class Claim:
    claim_id: str
    customer_id: str
    delivery_id: str
    reason: str
    amount: int
    declared_value: int
    opened_at: int
    deadline: int


def booking_record(value: object) -> Booking:
    """Adapt a booking; a route must join distinct locations in time order."""
    record = fields(value, "booking")
    booking_id = text(required(record, "booking_id"), "booking_id")
    customer_id = text(required(record, "customer_id"), "customer_id")
    origin = text(required(record, "origin"), "origin")
    destination = text(required(record, "destination"), "destination")
    weight = integer(required(record, "weight"), "weight", 1)
    ready_at = integer(required(record, "ready_at"), "ready_at")
    due_at = integer(required(record, "due_at"), "due_at")
    declared = integer(required(record, "declared_cents"), "declared_cents")
    if origin == destination:
        raise RecordError("origin and destination must differ")
    if ready_at >= due_at:
        raise RecordError("due_at must follow ready_at")
    return Booking(
        booking_id=booking_id,
        customer_id=customer_id,
        origin=origin,
        destination=destination,
        weight=weight,
        ready_at=ready_at,
        due_at=due_at,
        declared_cents=declared,
    )


def quote_record(value: object) -> Quote:
    """Adapt a quote whose expiry must follow its issue time."""
    record = fields(value, "quote")
    quote_id = text(required(record, "quote_id"), "quote_id")
    customer_id = text(required(record, "customer_id"), "customer_id")
    booking_id = text(required(record, "booking_id"), "booking_id")
    carrier_id = text(required(record, "carrier_id"), "carrier_id")
    cents = integer(required(record, "cents"), "cents", 1)
    issued = integer(required(record, "issued_at"), "issued_at")
    expires = integer(required(record, "expires_at"), "expires_at")
    capacity = integer(required(record, "capacity"), "capacity", 1)
    if issued >= expires:
        raise RecordError("expires_at must follow issued_at")
    return Quote(
        quote_id=quote_id,
        customer_id=customer_id,
        booking_id=booking_id,
        carrier_id=carrier_id,
        cents=cents,
        issued_at=issued,
        expires_at=expires,
        capacity=capacity,
    )


def vehicle_record(value: object) -> Vehicle:
    """Adapt a vehicle; its next inspection must follow the last one."""
    record = fields(value, "vehicle")
    vehicle_id = text(required(record, "vehicle_id"), "vehicle_id")
    carrier_id = text(required(record, "carrier_id"), "carrier_id")
    depot = text(required(record, "depot"), "depot")
    registration = text(required(record, "registration"), "registration")
    capacity = integer(required(record, "capacity"), "capacity", 1)
    available = integer(required(record, "available_at"), "available_at")
    inspected = integer(required(record, "inspected_at"), "inspected_at")
    due = integer(required(record, "inspection_due"), "inspection_due")
    return Vehicle(
        vehicle_id=vehicle_id,
        carrier_id=carrier_id,
        depot=depot,
        registration=registration,
        capacity=capacity,
        available_at=available,
        inspected_at=inspected,
        inspection_due=due,
    )


def consignment_record(value: object) -> Consignment:
    """Adapt a consignment due strictly after pickup."""
    record = fields(value, "consignment")
    consignment_id = text(required(record, "consignment_id"), "consignment_id")
    customer_id = text(required(record, "customer_id"), "customer_id")
    booking_id = text(required(record, "booking_id"), "booking_id")
    tracking_id = text(required(record, "tracking_id"), "tracking_id")
    pieces = integer(required(record, "pieces"), "pieces", 1)
    weight = integer(required(record, "weight"), "weight", 1)
    picked_up = integer(required(record, "picked_up_at"), "picked_up_at")
    deliver_by = integer(required(record, "deliver_by"), "deliver_by")
    if picked_up >= deliver_by:
        raise RecordError("deliver_by must follow pickup")
    return Consignment(
        consignment_id=consignment_id,
        customer_id=customer_id,
        booking_id=booking_id,
        tracking_id=tracking_id,
        pieces=pieces,
        weight=weight,
        picked_up_at=picked_up,
        deliver_by=deliver_by,
    )


def route_record(value: object) -> Route:
    """Adapt a carrier route between different locations in time order."""
    record = fields(value, "route")
    route_id = text(required(record, "route_id"), "route_id")
    carrier_id = text(required(record, "carrier_id"), "carrier_id")
    vehicle_id = text(required(record, "vehicle_id"), "vehicle_id")
    origin = text(required(record, "origin"), "origin")
    destination = text(required(record, "destination"), "destination")
    departure = integer(required(record, "depart_at"), "depart_at")
    arrival = integer(required(record, "arrive_at"), "arrive_at")
    capacity = integer(required(record, "capacity"), "capacity", 1)
    if origin == destination:
        raise RecordError("route endpoints must differ")
    return Route(
        route_id=route_id,
        carrier_id=carrier_id,
        vehicle_id=vehicle_id,
        origin=origin,
        destination=destination,
        depart_at=departure,
        arrive_at=arrival,
        capacity=capacity,
    )


def invoice_record(value: object) -> Invoice:
    """Adapt an invoice whose total is the subtotal plus tax."""
    record = fields(value, "invoice")
    invoice_id = text(required(record, "invoice_id"), "invoice_id")
    customer_id = text(required(record, "customer_id"), "customer_id")
    booking_id = text(required(record, "booking_id"), "booking_id")
    currency = text(required(record, "currency"), "currency")
    subtotal = integer(required(record, "subtotal"), "subtotal")
    tax = integer(required(record, "tax"), "tax")
    total = integer(required(record, "total"), "total")
    due_at = integer(required(record, "due_at"), "due_at")
    if total != subtotal + tax:
        raise RecordError("total must equal subtotal plus tax")
    return Invoice(
        invoice_id=invoice_id,
        customer_id=customer_id,
        booking_id=booking_id,
        currency=currency,
        subtotal=subtotal,
        tax=tax,
        total=total,
        due_at=due_at,
    )


def delivery_record(value: object) -> Delivery:
    """Adapt a delivery; damaged pieces cannot exceed received pieces."""
    record = fields(value, "delivery")
    delivery_id = text(required(record, "delivery_id"), "delivery_id")
    customer_id = required(record, "customer_id")
    consignment_id = text(required(record, "consignment_id"), "consignment_id")
    recipient = text(required(record, "recipient"), "recipient")
    received_at = integer(required(record, "received_at"), "received_at")
    pieces = integer(required(record, "pieces"), "pieces", 1)
    damaged = integer(required(record, "damaged"), "damaged")
    evidence_id = text(required(record, "evidence_id"), "evidence_id")
    if damaged > pieces:
        raise RecordError("damaged exceeds pieces")
    return Delivery(
        delivery_id=delivery_id,
        customer_id=customer_id,
        consignment_id=consignment_id,
        recipient=recipient,
        received_at=received_at,
        pieces=pieces,
        damaged=damaged,
        evidence_id=evidence_id,
    )


def claim_record(value: object) -> Claim:
    """Adapt a claim no larger than its declared value and before deadline."""
    record = fields(value, "claim")
    claim_id = text(required(record, "claim_id"), "claim_id")
    customer_id = text(required(record, "customer_id"), "customer_id")
    delivery_id = text(required(record, "delivery_id"), "delivery_id")
    reason = text(required(record, "reason"), "reason")
    amount = integer(required(record, "amount"), "amount", 1)
    declared = integer(required(record, "declared_value"), "declared_value", 1)
    opened = integer(required(record, "opened_at"), "opened_at")
    deadline = integer(required(record, "deadline"), "deadline")
    if amount > declared:
        raise RecordError("claim exceeds declared value")
    return Claim(
        claim_id=claim_id,
        customer_id=customer_id,
        delivery_id=delivery_id,
        reason=reason,
        amount=amount,
        declared_value=declared,
        opened_at=opened,
        deadline=deadline,
    )


def customer_can_view(customer_id: str, booking: Booking, invoice: Invoice) -> bool:
    """Only matching customers may view a booking and its invoice."""
    if booking.customer_id != customer_id:
        return False
    if invoice.customer_id != customer_id:
        return False
    return invoice.booking_id == booking.booking_id


def invoice_balance(invoice: Invoice, payments: list[int]) -> int:
    """Return the remaining amount; callers supply already validated payments."""
    return max(0, invoice.total - sum(payments))
