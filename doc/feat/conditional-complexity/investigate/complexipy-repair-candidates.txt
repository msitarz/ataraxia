def read_calls(log):
    calls = []
    for line in log.read_text().splitlines() if log.exists() else ():
        record: object = json.loads(line)
        calls.append(_uv_call_from_record(record))
    return tuple(calls)


def _uv_call_from_record(record: object) -> UvCall:
    if not isinstance(record, dict):
        raise ValueError("uv record must be a mapping")
    arguments: object = record.get("argv")
    environment: object = record.get("environment")
    if not isinstance(arguments, list) or not isinstance(environment, dict):
        raise ValueError("uv record must contain arguments and environment")
    argv: list[str] = []
    observed: dict[str, str | None] = {}
    for argument in arguments:
        if not isinstance(argument, str):
            raise ValueError("uv argument must be a string")
        argv.append(argument)
    for name, value in environment.items():
        if not isinstance(name, str) or not (
            value is None or isinstance(value, str)
        ):
            raise ValueError("uv environment must contain string names and values")
        observed[name] = value
    return UvCall(tuple(argv), observed)


def check_work(blocks):
    headings = _located_headings(blocks)
    findings = []
    title = next(
        (heading for heading in headings if heading.block.heading_level == 1), None
    )
    if title is None:
        findings.append(WorkDiagnostic(1, "required level-one Work name", "file start"))
    elif not title.text:
        findings.append(
            WorkDiagnostic(
                title.block.location.line_start,
                "nonempty Work name",
                "level-one heading",
            )
        )

    sections = {name: [] for name in _REQUIRED_SECTIONS}
    for heading in headings:
        if heading.block.heading_level == 2 and heading.text in sections:
            sections[heading.text].append(heading)
    findings.extend(_required_section_findings(blocks, sections))
    findings.extend(
        _criterion_findings(blocks, headings, sections["Acceptance criteria"])
    )
    return tuple(sorted(findings, key=lambda item: (item.line, item.rule, item.context)))


def _required_section_findings(blocks, sections):
    findings = []
    for name in _REQUIRED_SECTIONS:
        matches = sections[name]
        if not matches:
            findings.append(
                WorkDiagnostic(1, f"required top-level section '{name}'", "file start")
            )
        else:
            findings.extend(
                WorkDiagnostic(
                    duplicate.block.location.line_start,
                    f"duplicate top-level section '{name}'",
                    duplicate.block.text,
                )
                for duplicate in matches[1:]
            )
            for heading in matches:
                findings.extend(_nonempty_section(blocks, heading))
    return tuple(findings)


def check_root_owner(root, common_dir):
    for entry in root.iterdir():
        metadata = _metadata_for_entry(entry)
        if metadata is not None:
            _verify_metadata_owner(metadata, root, common_dir)


def _metadata_for_entry(entry):
    if entry.is_symlink():
        if entry.name.endswith(METADATA):
            raise ValueError(f"managed ownership metadata is a symlink: {entry}")
        return None
    if entry.name.endswith(METADATA):
        return entry
    if entry.is_dir():
        return entry / METADATA
    return None


def _verify_metadata_owner(metadata, root, common_dir):
    if metadata.is_symlink():
        raise ValueError(f"managed ownership metadata is a symlink: {metadata}")
    if metadata.is_file():
        record = json.loads(metadata.read_text())
        if record.get("common_dir") != common_dir:
            raise ValueError(f"managed root belongs to another checkout: {root}")
